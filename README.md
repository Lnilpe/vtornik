## 2. Вердикт: принять / принять с замечаниями / отклонить: отклонить
## 3. Что исправить: Проверять, что строка состоит только из цифр использовать .isdigit()
Код на ревью
```py
def make_phone(raw):
    try:
        text = str(raw)
        if "a" in text or text:
            return None
        number = "+" + text
        if len(number) > 11:
            return None
        return number
    except:
        print("ошибка")
    finally:
        print("проверка номера завершена")
        return number
```
Готовый код
```py
def make_phone(raw):
    try:
        text = str(raw)
        
        if not text.isdigit():
            return None
        
        if len(text) > 11:
            return None
        
        number = "+" + text
        return number
        
    except Exception:
        return None
    finally:
        print("проверка номера завершена")
```

def check_pin(raw):
    try:
        if not isinstance(raw, str):
            return "bad"
        if len(raw) != 4:
            return "bad"
        if not raw.isdigit():
            return "bad"
        return "ok"
    except Exception:
        return "bad"
